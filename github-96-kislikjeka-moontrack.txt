import { Transaction } from '@/types/transaction';

/**
 * Checks if transaction is a USDC bridge deposit (not a swap).
 * A bridge deposit has:
 * - USDC as input token
 * - Destination chain different from source chain
 * - No ETH involved in the transaction
 * - Proper bridge contract address in the transaction
 */
export function isUSDCBridgeDeposit(tx: Transaction): boolean {
  // Check if transaction involves USDC
  if (!tx.tokens?.some(t => t.symbol === 'USDC' && t.direction === 'in')) {
    return false;
  }
  
  // Verify it's not a swap (no output token conversion)
  if (tx.tokens?.some(t => t.direction === 'out' && t.symbol !== 'USDC')) {
    return false;
  }
  
  // Check for bridge-specific contract address
  const bridgeContract = getUSDCBridgeContractAddress();
  if (tx.to !== bridgeContract && tx.from !== bridgeContract) {
    return false;
  }
  
  // Ensure no ETH dust return is present (ETH should not be in the transaction)
  if (tx.tokens?.some(t => t.symbol === 'ETH')) {
    return false;
  }
  
  return true;
}

/**
 * Checks if transaction is an ETH→USDC swap.
 * A swap has:
 * - ETH as input token
 * - USDC as output token
 * - No bridge contract involvement
 */
export function isETHToUSDCSwap(tx: Transaction): boolean {
  // Must have ETH input
  const hasETHInput = tx.tokens?.some(t => t.symbol === 'ETH' && t.direction === 'in');
  if (!hasETHInput) {
    return false;
  }
  
  // Must have USDC output
  const hasUSDCOutput = tx.tokens?.some(t => t.symbol === 'USDC' && t.direction === 'out');
  if (!hasUSDCOutput) {
    return false;
  }
  
  // Must not involve bridge contract
  const bridgeContract = getUSDCBridgeContractAddress();
  if (tx.to === bridgeContract || tx.from === bridgeContract) {
    return false;
  }
  
  return true;
}
